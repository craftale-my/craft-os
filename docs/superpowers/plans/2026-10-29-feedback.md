# Staff Feedback Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let every signed-in staff member send feedback in four categories from a floating button, the sidebar, or an error screen, and give the owner a private Inbox to set a status and reply.

**Architecture:** One new `feedback` table that nobody writes directly: `submit_feedback()` and `reply_feedback()` are security definer functions that decide whether to keep the submitter (anonymous Workplace feedback stores none) and insert the `notifications` rows the existing bell already shows. Photos go to a private bucket and are read through signed URLs. The front end is a self-contained `src/features/feedback/` module: a pure validator, a thin Supabase API layer, a tiny event bus so the error screen and the floating button open the same sheet, and one `/feedback` page with `My Feedback` and an owner-only `Inbox`.

**Tech Stack:** React 19 + TypeScript + Vite, Tailwind, react-router-dom 7, lucide-react, Supabase (Postgres + RLS + Storage), vitest + @testing-library/react (jsdom).

Spec: `docs/superpowers/specs/2026-10-29-feedback-design.md`.

## Global Constraints

- All UI labels, copy, placeholders and error strings are **English**. Code comments may be Chinese (repo convention).
- Category values and labels, exactly: `product_quality` → `Product Quality`, `product_idea` → `Product Idea`, `system` → `App / System Issue`, `workplace` → `Workplace / Complaint`.
- Status values and labels, exactly: `new` → `New`, `in_review` → `In review`, `resolved` → `Resolved`.
- Limits: message 1–2000 characters after trim; at most 3 photos; each photo an image under 5 MB (5242880 bytes); owner reply ≤ 2000 characters.
- Only `workplace` may be anonymous. Anonymous rows store `submitted_by = null`, no photos, and `context` reduced to `appVersion` only.
- Only the owner reads the Inbox. Front end: `staff.system_role === 'owner'`. SQL: the existing `current_is_owner()` (`migration-2026-07-06-rbac.sql:98`).
- Feedback notifications leave `subject_staff_id` and `subject_date` null (avoids the `notifications` unique key at `migration-2026-09-02-late-reason-notifications.sql:972`).

```sql
-- ============================================================================
-- VERIFY feedback (2026-10-29). 整段在一个事务里跑,最后 rollback,不留数据。
-- 任一检查失败都会 raise 'VERIFY FAIL: …',整个请求报错。
-- 需要:1 个在职 owner + 2 个在职 staff(dev 和 prod 都满足)。
-- ============================================================================
begin;

create temp table _v on commit drop as
select
  (select id from staff where system_role = 'owner' and is_active order by created_at limit 1)          as owner_id,
  (select id from staff where system_role = 'staff' and is_active order by created_at limit 1)          as staff_id,
  (select id from staff where system_role = 'staff' and is_active order by created_at offset 1 limit 1) as other_id;

do $$ begin
  if (select owner_id is null or staff_id is null or other_id is null from _v) then
    raise exception 'VERIFY SETUP: need 1 active owner and 2 active staff';
  end if;
end $$;

-- ---- 1. 以 staff 身份提交:两条实名 + 一条匿名 ---------------------------------
select set_config('request.jwt.claims', json_build_object('sub', staff_id, 'role', 'authenticated')::text, true) from _v;
set local role authenticated;

select submit_feedback('11111111-1111-4111-8111-111111111111', 'product_quality', 'Latte too sour today',
  'Cheras (Taman Connaught)', '/profile', '{"userAgent":"verify","appVersion":"v"}', '{}', false);
select submit_feedback('22222222-2222-4222-8222-222222222222', 'system', 'Leave page crashed',
  null, '/hr/leave', '{"appVersion":"v"}', '{}', false);
select submit_feedback('33333333-3333-4333-8333-333333333333', 'workplace', 'Anonymous complaint',
  null, '/profile', '{"userAgent":"verify","appVersion":"v","errorMessage":"x"}', '{}', true);

-- ---- 2. 非 Workplace 匿名被拒 -------------------------------------------------
do $$ begin
  begin
    perform submit_feedback('44444444-4444-4444-8444-444444444444', 'system', 'x', null, null, '{}', '{}', true);
    raise exception 'VERIFY FAIL: anonymous system feedback was accepted';
  exception when others then
    if sqlerrm like 'VERIFY FAIL%' then raise; end if;
  end;
end $$;

-- ---- 3. staff 只看得到自己的两条实名 ------------------------------------------
do $$ begin
  if (select count(*) from feedback where id in (
        '11111111-1111-4111-8111-111111111111',
        '22222222-2222-4222-8222-222222222222',
        '33333333-3333-4333-8333-333333333333')) <> 2 then
    raise exception 'VERIFY FAIL: staff should see exactly own 2 named rows';
  end if;
end $$;

-- ---- 4. staff 不能回复 --------------------------------------------------------
do $$ begin
  begin
    perform reply_feedback('11111111-1111-4111-8111-111111111111', 'resolved', 'hi');
    raise exception 'VERIFY FAIL: staff could reply';
  exception when others then
    if sqlerrm like 'VERIFY FAIL%' then raise; end if;
  end;
end $$;

-- ---- 5. 直接 insert 被 RLS 挡住 ----------------------------------------------
do $$ begin
  begin
    insert into feedback (id, category, message) values ('55555555-5555-4555-8555-555555555555', 'system', 'direct');
    raise exception 'VERIFY FAIL: direct insert allowed';
  exception when others then
    if sqlerrm like 'VERIFY FAIL%' then raise; end if;
  end;
end $$;

-- ---- 6. 另一个 staff 一条都看不到 ---------------------------------------------
reset role;
select set_config('request.jwt.claims', json_build_object('sub', other_id, 'role', 'authenticated')::text, true) from _v;
set local role authenticated;

do $$ begin
  if (select count(*) from feedback where id in (
        '11111111-1111-4111-8111-111111111111',
        '22222222-2222-4222-8222-222222222222',
        '33333333-3333-4333-8333-333333333333')) <> 0 then
    raise exception 'VERIFY FAIL: other staff can see feedback';
  end if;
end $$;

-- ---- 7. owner 看得到全部三条;匿名那条已被剥干净 -------------------------------
reset role;
select set_config('request.jwt.claims', json_build_object('sub', owner_id, 'role', 'authenticated')::text, true) from _v;
set local role authenticated;

do $$
declare r feedback%rowtype;
begin
  if (select count(*) from feedback where id in (
        '11111111-1111-4111-8111-111111111111',
        '22222222-2222-4222-8222-222222222222',
        '33333333-3333-4333-8333-333333333333')) <> 3 then
    raise exception 'VERIFY FAIL: owner should see all 3';
  end if;
  select * into r from feedback where id = '33333333-3333-4333-8333-333333333333';
  if r.submitted_by is not null then raise exception 'VERIFY FAIL: anonymous row kept submitted_by'; end if;
  if r.context <> '{"appVersion":"v"}'::jsonb then raise exception 'VERIFY FAIL: anonymous context not stripped: %', r.context; end if;
  if cardinality(r.photo_paths) <> 0 then raise exception 'VERIFY FAIL: anonymous row has photos'; end if;
end $$;

-- ---- 8. owner 回复实名和匿名各一条 --------------------------------------------
select reply_feedback('11111111-1111-4111-8111-111111111111', 'resolved', 'Thanks, fixed the grinder');
select reply_feedback('33333333-3333-4333-8333-333333333333', 'in_review', 'Looking into it');

-- ---- 9. 通知与存储(以 postgres 身份查,绕过 RLS) -------------------------------
reset role;

do $$
declare v record;
begin
  select * into v from _v;

  if (select count(*) from notifications
       where recipient_id = v.owner_id and event_type = 'feedback_submitted'
         and link in ('/feedback?tab=inbox&id=11111111-1111-4111-8111-111111111111',
                      '/feedback?tab=inbox&id=22222222-2222-4222-8222-222222222222',
                      '/feedback?tab=inbox&id=33333333-3333-4333-8333-333333333333')) <> 3 then
    raise exception 'VERIFY FAIL: owner should get 3 feedback_submitted notifications (same staff, same day)';
  end if;

  if (select count(*) from notifications
       where recipient_id = v.staff_id and event_type = 'feedback_updated'
         and link = '/feedback?id=11111111-1111-4111-8111-111111111111') <> 1 then
    raise exception 'VERIFY FAIL: named submitter should get 1 feedback_updated notification';
  end if;

  if exists (select 1 from notifications
              where event_type = 'feedback_updated'
                and link = '/feedback?id=33333333-3333-4333-8333-333333333333') then
    raise exception 'VERIFY FAIL: anonymous feedback produced a reply notification';
  end if;

  if (select status <> 'resolved' or owner_reply <> 'Thanks, fixed the grinder' or replied_at is null
        from feedback where id = '11111111-1111-4111-8111-111111111111') then
    raise exception 'VERIFY FAIL: reply not saved';
  end if;

  if (select public from storage.buckets where id = 'feedback-photos') is distinct from false then
    raise exception 'VERIFY FAIL: feedback-photos bucket missing or public';
  end if;

  if (select count(*) from pg_policies
       where schemaname = 'storage' and tablename = 'objects'
         and policyname in ('feedback_photos_insert', 'feedback_photos_select', 'feedback_photos_delete_orphan')) <> 3 then
    raise exception 'VERIFY FAIL: feedback storage policies missing';
  end if;
end $$;

rollback;

select 'VERIFY PASSED' as result;
```

```sql
-- ============================================================================
-- Feedback (2026-10-29). Idempotent.
-- Spec: docs/superpowers/specs/2026-10-29-feedback-design.md
--
-- 写入只走两个 security definer 函数:表上刻意不建 insert/update/delete policy。
-- 匿名(只限 workplace)时由服务器丢掉提交人、照片和 userAgent,
-- 而不是信任客户端传来的值。
-- ============================================================================

create table if not exists feedback (
  id            uuid primary key,              -- 客户端生成,照片先按 id 上传
  category      text not null,
  message       text not null,
  branch        text,
  page_path     text,
  context       jsonb not null default '{}'::jsonb,
  photo_paths   text[] not null default '{}',
  is_anonymous  boolean not null default false,
  submitted_by  uuid references staff(id) on delete set null,
  status        text not null default 'new',
  owner_reply   text,
  replied_at    timestamptz,
  created_at    timestamptz not null default now(),
  updated_at    timestamptz not null default now()
);

alter table feedback drop constraint if exists feedback_category_check;
alter table feedback add  constraint feedback_category_check
  check (category in ('product_quality', 'product_idea', 'system', 'workplace'));

alter table feedback drop constraint if exists feedback_message_check;
alter table feedback add  constraint feedback_message_check
  check (char_length(btrim(message)) between 1 and 2000);

alter table feedback drop constraint if exists feedback_status_check;
alter table feedback add  constraint feedback_status_check
  check (status in ('new', 'in_review', 'resolved'));

alter table feedback drop constraint if exists feedback_reply_check;
alter table feedback add  constraint feedback_reply_check
  check (owner_reply is null or char_length(owner_reply) <= 2000);

alter table feedback drop constraint if exists feedback_photos_check;
alter table feedback add  constraint feedback_photos_check
  check (cardinality(photo_paths) <= 3);

alter table feedback drop constraint if exists feedback_anonymous_only_workplace;
alter table feedback add  constraint feedback_anonymous_only_workplace
  check (not is_anonymous or category = 'workplace');

alter table feedback drop constraint if exists feedback_anonymous_no_submitter;
alter table feedback add  constraint feedback_anonymous_no_submitter
  check (not is_anonymous or submitted_by is null);

alter table feedback drop constraint if exists feedback_anonymous_no_photos;
alter table feedback add  constraint feedback_anonymous_no_photos
  check (not is_anonymous or cardinality(photo_paths) = 0);

create index if not exists feedback_status_idx    on feedback (status, created_at desc);
create index if not exists feedback_submitter_idx on feedback (submitted_by, created_at desc);

alter table feedback enable row level security;

drop policy if exists "feedback_select" on feedback;
create policy "feedback_select" on feedback for select
  using (submitted_by = auth.uid() or current_is_owner());

-- ---------------------------------------------------------------------------
-- submit_feedback
-- ---------------------------------------------------------------------------
create or replace function submit_feedback(
  p_id          uuid,
  p_category    text,
  p_message     text,
  p_branch      text,
  p_page_path   text,
  p_context     jsonb,
  p_photo_paths text[],
  p_anonymous   boolean
) returns uuid
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  v_uid   uuid    := auth.uid();
  v_anon  boolean := coalesce(p_anonymous, false);
  v_label text;
begin
  if v_uid is null then
    raise exception 'Not signed in.' using errcode = '42501';
  end if;
  if not exists (select 1 from staff where id = v_uid and is_active) then
    raise exception 'Only active staff can send feedback.' using errcode = '42501';
  end if;
  if v_anon and p_category is distinct from 'workplace' then
    raise exception 'Only Workplace feedback can be anonymous.' using errcode = '22023';
  end if;
  -- 照片必须放在这条反馈自己的文件夹下,不能指向别人的
  if exists (select 1 from unnest(coalesce(p_photo_paths, '{}')) as p(path)
              where p.path not like p_id::text || '/%') then
    raise exception 'Invalid photo path.' using errcode = '22023';
  end if;

  insert into feedback (id, category, message, branch, page_path, context, photo_paths, is_anonymous, submitted_by)
  values (
    p_id,
    p_category,
    btrim(p_message),
    nullif(btrim(coalesce(p_branch, '')), ''),
    p_page_path,
    case when v_anon
      then jsonb_strip_nulls(jsonb_build_object('appVersion', p_context -> 'appVersion'))
      else coalesce(p_context, '{}'::jsonb) end,
    case when v_anon then '{}'::text[] else coalesce(p_photo_paths, '{}') end,
    v_anon,
    case when v_anon then null else v_uid end
  );

  v_label := case p_category
    when 'product_quality' then 'Product Quality'
    when 'product_idea'    then 'Product Idea'
    when 'system'          then 'App / System Issue'
    else                        'Workplace / Complaint'
  end;

  -- subject_staff_id / subject_date 留 null:notifications 的 unique 键里 NULL 互不相等,
  -- 同一人同一天的多条反馈不会互相顶掉。
  insert into notifications (recipient_id, event_type, title, body, link)
  select s.id,
         'feedback_submitted',
         'New feedback: ' || v_label,
         left(btrim(p_message), 120),
         '/feedback?tab=inbox&id=' || p_id::text
    from staff s
   where s.system_role = 'owner'
     and s.is_active
     and (v_anon or s.id <> v_uid);   -- owner 自己实名提交时不通知自己

  return p_id;
end;
$$;

revoke execute on function submit_feedback(uuid, text, text, text, text, jsonb, text[], boolean) from public;
revoke execute on function submit_feedback(uuid, text, text, text, text, jsonb, text[], boolean) from anon;
grant  execute on function submit_feedback(uuid, text, text, text, text, jsonb, text[], boolean) to authenticated;

-- ---------------------------------------------------------------------------
-- reply_feedback
-- ---------------------------------------------------------------------------
create or replace function reply_feedback(p_id uuid, p_status text, p_reply text)
returns void
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  v_row           feedback%rowtype;
  v_reply         text := nullif(btrim(coalesce(p_reply, '')), '');
  v_reply_changed boolean;
  v_status_label  text;
begin
  if not current_is_owner() then
    raise exception 'Only the owner can reply to feedback.' using errcode = '42501';
  end if;

  select * into v_row from feedback where id = p_id for update;
  if not found then
    raise exception 'Feedback not found.' using errcode = 'P0002';
  end if;

  -- 空回复 = 不动原回复(只改状态)
  v_reply_changed := v_reply is not null and v_reply is distinct from v_row.owner_reply;

  update feedback set
    status      = p_status,
    owner_reply = case when v_reply_changed then v_reply else owner_reply end,
    replied_at  = case when v_reply_changed then now()   else replied_at  end,
    updated_at  = now()
  where id = p_id;

  if v_row.submitted_by is not null and (v_reply_changed or p_status <> v_row.status) then
    v_status_label := case p_status when 'new' then 'New' when 'in_review' then 'In review' else 'Resolved' end;
    insert into notifications (recipient_id, event_type, title, body, link)
    values (
      v_row.submitted_by,
      'feedback_updated',
      'Your feedback was updated',
      v_status_label || case when v_reply_changed then ' — ' || left(v_reply, 120) else '' end,
      '/feedback?id=' || p_id::text
    );
  end if;
end;
$$;

revoke execute on function reply_feedback(uuid, text, text) from public;
revoke execute on function reply_feedback(uuid, text, text) from anon;
grant  execute on function reply_feedback(uuid, text, text) to authenticated;

-- ---------------------------------------------------------------------------
-- Photos: private bucket, signed URLs only
-- ---------------------------------------------------------------------------
insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
values ('feedback-photos', 'feedback-photos', false, 5242880, array['image/*'])
on conflict (id) do update
  set public = false,
      file_size_limit = excluded.file_size_limit,
      allowed_mime_types = excluded.allowed_mime_types;

-- 上传:路径第一段必须是 uuid。这里不查「那条反馈是否已存在」:policy 里的子查询
-- 受调用者自己的 RLS 限制,看不到别人的反馈,查了也拦不住。往别人的文件夹里多塞
-- 一张图也无害 —— photo_paths 在提交那一刻就定死了,多出来的文件永远不会被引用。
drop policy if exists "feedback_photos_insert" on storage.objects;
create policy "feedback_photos_insert" on storage.objects for insert to authenticated
  with check (
    bucket_id = 'feedback-photos'
    and (storage.foldername(name))[1] ~ '^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$'
  );

drop policy if exists "feedback_photos_select" on storage.objects;
create policy "feedback_photos_select" on storage.objects for select to authenticated
  using (
    bucket_id = 'feedback-photos'
    and (
      public.current_is_owner()
      or exists (select 1 from public.feedback f
                  where f.id::text = (storage.foldername(name))[1]
                    and f.submitted_by = auth.uid())
    )
  );

-- 提交失败时的清理:只能删自己传的、还没挂到任何反馈上的照片
drop policy if exists "feedback_photos_delete_orphan" on storage.objects;
create policy "feedback_photos_delete_orphan" on storage.objects for delete to authenticated
  using (
    bucket_id = 'feedback-photos'
    and owner_id = auth.uid()::text
    and not exists (select 1 from public.feedback f where f.id::text = (storage.foldername(name))[1])
  );
```

```ts
import { MAX_MESSAGE, MAX_PHOTOS, MAX_PHOTO_BYTES } from './types'
import type { FeedbackCategory } from './types'

type PhotoLike = { size: number; type: string }

export interface FeedbackDraft {
  category: FeedbackCategory | null
  message: string
  photos: PhotoLike[]
  anonymous: boolean
}

export function isAcceptablePhoto(f: PhotoLike): boolean {
  return f.type.startsWith('image/') && f.size <= MAX_PHOTO_BYTES
}

/** 提交前的校验。服务器端约束是同一套规则的兜底,不是替代。 */
export function validateFeedbackDraft(d: FeedbackDraft): string[] {
  const errors: string[] = []
  if (!d.category) errors.push('Choose a category.')
  const text = d.message.trim()
  if (text.length === 0) errors.push('Write a message.')
  if (text.length > MAX_MESSAGE) errors.push(`Message must be ${MAX_MESSAGE} characters or fewer.`)
  if (d.photos.length > MAX_PHOTOS) errors.push(`Add up to ${MAX_PHOTOS} photos.`)
  if (d.photos.some(p => !isAcceptablePhoto(p))) errors.push('Photos must be images under 5 MB.')
  if (d.anonymous && d.category !== 'workplace') errors.push('Only Workplace feedback can be anonymous.')
  if (d.anonymous && d.photos.length > 0) errors.push("Photos can't be attached to anonymous feedback.")
  return errors
}
```

```ts
import { supabase } from '../../shared/lib/supabase'
import type { FeedbackCategory, FeedbackContext, FeedbackRow, FeedbackStatus } from './types'

const BUCKET = 'feedback-photos'

export function feedbackContext(errorMessage?: string): FeedbackContext {
  return {
    userAgent: navigator.userAgent,
    appVersion: __APP_VERSION__,
    ...(errorMessage ? { errorMessage: errorMessage.slice(0, 500) } : {}),
  }
}

export interface SubmitInput {
  category: FeedbackCategory
  message: string
  branch: string | null
  anonymous: boolean
  photos: File[]
  pagePath: string
  context: FeedbackContext
}

export type SubmitResult = { ok: true; id: string } | { ok: false; reason: string }

/** 先传照片再写记录;任何一步失败都把已传的照片删掉,不留孤儿档案。 */
export async function submitFeedback(input: SubmitInput): Promise<SubmitResult> {
  const id = crypto.randomUUID()
  // 匿名不传照片:storage.objects 会记下上传人,照片会暴露身份
  const photos = input.anonymous ? [] : input.photos
  const paths: string[] = []

  for (const [i, file] of photos.entries()) {
    const ext = (file.name.split('.').pop() || 'jpg').toLowerCase()
    const path = `${id}/${i + 1}.${ext}`
    const { error } = await supabase.storage.from(BUCKET).upload(path, file, { contentType: file.type })
    if (error) {
      await removePhotos(paths)
      return { ok: false, reason: "Couldn't upload photo. Try again." }
    }
    paths.push(path)
  }

  const { error } = await supabase.rpc('submit_feedback', {
    p_id: id,
    p_category: input.category,
    p_message: input.message,
    p_branch: input.branch,
    p_page_path: input.pagePath,
    p_context: input.context,
    p_photo_paths: paths,
    p_anonymous: input.anonymous,
  })
  if (error) {
    await removePhotos(paths)
    return { ok: false, reason: error.message }
  }
  return { ok: true, id }
}

async function removePhotos(paths: string[]) {
  if (paths.length === 0) return
  const { error } = await supabase.storage.from(BUCKET).remove(paths)
  if (error) console.error(`[feedback] orphaned photos not removed: ${paths.join(', ')}`, error.message)
}

export async function replyFeedback(
  id: string,
  status: FeedbackStatus,
  reply: string,
): Promise<{ ok: true } | { ok: false; reason: string }> {
  const { error } = await supabase.rpc('reply_feedback', { p_id: id, p_status: status, p_reply: reply })
  return error ? { ok: false, reason: error.message } : { ok: true }
}

export async function listMyFeedback(staffId: string): Promise<FeedbackRow[]> {
  const { data, error } = await supabase
    .from('feedback')
    .select('*')
    .eq('submitted_by', staffId)
    .order('created_at', { ascending: false })
  if (error) throw new Error(error.message)
  return ((data ?? []) as FeedbackRow[]).map(f => ({ ...f, submitter: null }))
}

/** RLS 只让 owner 读到全部,这里不再加过滤。 */
export async function listInbox(): Promise<FeedbackRow[]> {
  const { data, error } = await supabase
    .from('feedback')
    .select('*, submitter:staff!feedback_submitted_by_fkey(name)')
    .order('created_at', { ascending: false })
    .limit(200)
  if (error) throw new Error(error.message)
  return (data ?? []) as FeedbackRow[]
}

export async function countNewFeedback(): Promise<number> {
  const { count, error } = await supabase
    .from('feedback')
    .select('id', { count: 'exact', head: true })
    .eq('status', 'new')
  return error ? 0 : count ?? 0
}

/** 私有 bucket:一小时有效的签名链接。 */
export async function photoUrls(paths: string[]): Promise<string[]> {
  if (paths.length === 0) return []
  const { data, error } = await supabase.storage.from(BUCKET).createSignedUrls(paths, 3600)
  if (error || !data) return []
  return data.map(d => d.signedUrl).filter((u): u is string => !!u)
}
```

```ts
export function openFeedback(prefill: FeedbackPrefill = {}) {
  listeners.forEach(l => l(prefill))
}

export function useFeedbackRequests(onOpen: Listener) {
  const ref = useRef(onOpen)
  useEffect(() => { ref.current = onOpen })
  useEffect(() => {
    const l: Listener = p => ref.current(p)
    listeners.add(l)
    return () => { listeners.delete(l) }
  }, [])
}
```

```tsx
import { useEffect, useMemo, useState } from 'react'
import { useLocation } from 'react-router-dom'
import { X } from 'lucide-react'
import { useAuth } from '../auth/AuthContext'
import { BRANCHES } from '../../shared/types'
import { submitFeedback, feedbackContext } from './api'
import { CATEGORIES, MAX_MESSAGE, MAX_PHOTOS } from './types'
import type { FeedbackCategory } from './types'
import { validateFeedbackDraft, isAcceptablePhoto } from './validate'
import type { FeedbackPrefill } from './feedbackBus'

export function FeedbackSheet({ prefill, onClose, onSent }: {
  prefill: FeedbackPrefill
  onClose: () => void
  onSent: () => void
}) {
  const { staff } = useAuth()
  const { pathname } = useLocation()

  const [category, setCategory] = useState<FeedbackCategory | null>(prefill.category ?? null)
  const [message, setMessage] = useState('')
  const [photos, setPhotos] = useState<File[]>([])
  const [branch, setBranch] = useState(staff?.branch ?? '')
  const [anonymous, setAnonymous] = useState(false)
  const [submitting, setSubmitting] = useState(false)
  const [error, setError] = useState<string | null>(null)

  // 员工资料里的分店不在清单里时也要能选到
  const branchOptions = staff?.branch && !BRANCHES.includes(staff.branch) ? [staff.branch, ...BRANCHES] : BRANCHES

  const previews = useMemo(() => photos.map(f => URL.createObjectURL(f)), [photos])
  useEffect(() => () => previews.forEach(u => URL.revokeObjectURL(u)), [previews])

  function pickCategory(c: FeedbackCategory) {
    setCategory(c)
    if (c !== 'workplace') setAnonymous(false)
  }

  function toggleAnonymous(on: boolean) {
    setAnonymous(on)
    if (on) setPhotos([])
  }

  function addPhotos(files: FileList | null) {
    if (!files) return
    const picked = Array.from(files)
    const good = picked.filter(isAcceptablePhoto)
    const next = [...photos, ...good]
    if (good.length < picked.length) setError('Photos must be images under 5 MB.')
    else if (next.length > MAX_PHOTOS) setError(`Add up to ${MAX_PHOTOS} photos.`)
    else setError(null)
    setPhotos(next.slice(0, MAX_PHOTOS))
  }

  async function send() {
    const errors = validateFeedbackDraft({ category, message, photos, anonymous })
    if (errors.length > 0 || !category) { setError(errors[0] ?? 'Choose a category.'); return }
    setSubmitting(true)
    setError(null)
    const res = await submitFeedback({
      category,
      message: message.trim(),
      branch: branch || null,
      anonymous,
      photos,
      pagePath: pathname,
      context: feedbackContext(prefill.errorMessage),
    })
    setSubmitting(false)
    if (res.ok) onSent()
    else setError(res.reason)
  }

  return (
    <div className="fixed inset-0 z-[70] bg-black/40 flex items-end sm:items-center justify-center" onClick={onClose}>
      <div
        role="dialog"
        aria-label="Send feedback"
        className="bg-white w-full sm:max-w-lg rounded-t-2xl sm:rounded-2xl max-h-[90vh] overflow-y-auto p-5 pb-[calc(1.25rem+env(safe-area-inset-bottom))]"
        onClick={e => e.stopPropagation()}
      >
        <div className="flex items-center justify-between mb-4">
          <h2 className="text-lg font-semibold text-brown-dark">Send feedback</h2>
          <button onClick={onClose} aria-label="Close" className="text-brown-muted hover:text-brown-dark p-1">
            <X size={20} />
          </button>
        </div>

        <p className="text-sm font-medium text-brown-dark mb-2">Category</p>
        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 mb-4">
          {CATEGORIES.map(c => (
            <button
              key={c.value}
              type="button"
              aria-pressed={category === c.value}
              onClick={() => pickCategory(c.value)}
              className={`text-left rounded-xl border px-3 py-2.5 transition-colors ${
                category === c.value ? 'border-[#8B6344] bg-[#F5F0E8]' : 'border-gray-200 hover:border-[#D4C4B0]'
              }`}
            >
              <span className="block text-sm font-semibold text-brown-dark">{c.label}</span>
              <span className="block text-xs text-brown-muted mt-0.5">{c.hint}</span>
            </button>
          ))}
        </div>

        <label className="block text-sm font-medium text-brown-dark mb-1" htmlFor="feedback-message">Message</label>
        <textarea
          id="feedback-message"
          value={message}
          onChange={e => setMessage(e.target.value)}
          rows={5}
          maxLength={MAX_MESSAGE}

          className="w-full rounded-xl border border-gray-200 px-3 py-2 text-sm focus:outline-none focus:border-[#8B6344]"
        />
        <p className="text-xs text-brown-muted text-right mb-3">{message.length}/{MAX_MESSAGE}</p>

        {anonymous ? (
          <p className="text-xs text-brown-muted mb-3">Photos can't be attached to anonymous feedback.</p>
        ) : (
          <div className="mb-3">
            <p className="text-sm font-medium text-brown-dark mb-1">Photos (optional)</p>
            <div className="flex gap-2 flex-wrap">
              {previews.map((url, i) => (

                  <button
                    type="button"
                    aria-label={`Remove photo ${i + 1}`}
                    onClick={() => setPhotos(photos.filter((_, j) => j !== i))}
                    className="absolute -top-1.5 -right-1.5 bg-[#3D2410] text-white rounded-full w-5 h-5 flex items-center justify-center"
                  >
                    <X size={12} />
                  </button>
                </div>

                  />
                </label>
              )}
            </div>
          </div>

    </div>
  )
}
```

```ts
import { useEffect, useState } from 'react'
import { countNewFeedback } from './api'

/**
 * Owner 侧栏徽标。不订阅 realtime:SidebarContent 在手机抽屉打开时会挂第二份
 * (见 Sidebar.tsx 铃铛旁的注释),订阅会开两条频道。换页时重查就够了。
 */
export function useNewFeedbackCount(enabled: boolean, refreshKey: string): number {
  const [count, setCount] = useState(0)
  useEffect(() => {
    if (!enabled) { setCount(0); return }
    let live = true
    countNewFeedback().then(n => { if (live) setCount(n) })
    return () => { live = false }
  }, [enabled, refreshKey])
  return count
}
```

- [ ] **Step 4: Write `FeedbackPage.tsx`**

```tsx
import { useEffect, useState } from 'react'
import { useSearchParams } from 'react-router-dom'
import { useAuth } from '../auth/AuthContext'
import { listInbox, listMyFeedback } from './api'
import { CATEGORIES, CATEGORY_LABELS, STATUS_LABELS } from './types'
import type { FeedbackCategory, FeedbackRow, FeedbackStatus } from './types'
import { FeedbackDetail, formatWhen } from './FeedbackDetail'
import { openFeedback } from './feedbackBus'

type StatusFilter = 'open' | 'all' | FeedbackStatus

export default function FeedbackPage() {
  const { staff } = useAuth()
  const owner = staff?.system_role === 'owner'
  const [params, setParams] = useSearchParams()
  const tab: 'mine' | 'inbox' = owner && params.get('tab') === 'inbox' ? 'inbox' : 'mine'
  const selectedId = params.get('id')

  const [items, setItems] = useState<FeedbackRow[] | null>(null)
  const [error, setError] = useState<string | null>(null)
  const [reloadKey, setReloadKey] = useState(0)
  const [statusFilter, setStatusFilter] = useState<StatusFilter>('open')
  const [categoryFilter, setCategoryFilter] = useState<'all' | FeedbackCategory>('all')

  useEffect(() => {
    if (!staff) return
    let live = true
    setItems(null)
    setError(null)
    const load = tab === 'inbox' ? listInbox() : listMyFeedback(staff.id)
    load.then(
      rows => { if (live) setItems(rows) },
      (e: Error) => { if (live) setError(e.message) },
    )
    return () => { live = false }
  }, [staff, tab, reloadKey])

  function go(next: { tab?: 'mine' | 'inbox'; id?: string | null }) {
    const p = new URLSearchParams()
    const t = next.tab ?? tab
    if (t === 'inbox') p.set('tab', 'inbox')
    const id = next.id === undefined ? selectedId : next.id
    if (id) p.set('id', id)
    setParams(p)
  }

  const selected = selectedId ? items?.find(f => f.id === selectedId) ?? null : null

  const visible = (items ?? []).filter(f => {
    if (tab !== 'inbox') return true
    if (statusFilter === 'open' && f.status === 'resolved') return false
    if (statusFilter !== 'open' && statusFilter !== 'all' && f.status !== statusFilter) return false
    if (categoryFilter !== 'all' && f.category !== categoryFilter) return false
    return true
  })

  return (
    <div className="max-w-2xl mx-auto px-4 py-6 pb-24">
      <div className="flex items-center justify-between mb-4">
        <h1 className="text-2xl font-display font-bold text-brown-dark">Feedback</h1>
        <button
          onClick={() => openFeedback()}
          className="text-sm bg-brown-btn hover:bg-brown-btn-hover text-white px-3 py-2 rounded-lg transition-colors"
        >
          New feedback
        </button>
      </div>

      {owner && (
        <div className="flex gap-2 mb-4">
          {(['mine', 'inbox'] as const).map(t => (
            <button
              key={t}
              onClick={() => go({ tab: t, id: null })}
              aria-pressed={tab === t}
              className={`text-sm px-3 py-1.5 rounded-full ${tab === t ? 'bg-[#4A2E1A] text-[#F5F0E8]' : 'bg-white text-brown-dark'}`}
            >
              {t === 'mine' ? 'My Feedback' : 'Inbox'}
            </button>
          ))}
        </div>
      )}

      {error && <p role="alert" className="text-sm text-red-600">{error}</p>}
      {!error && items === null && <p className="text-sm text-brown-muted">Loading…</p>}

      {items && selected && (
        <FeedbackDetail
          key={selected.id}
          item={selected}
          ownerView={tab === 'inbox'}
          onBack={() => go({ id: null })}
          onSaved={() => setReloadKey(k => k + 1)}
        />
      )}

      {items && selectedId && !selected && (
        <p className="text-sm text-brown-muted">This feedback isn't available.</p>
      )}

      {items && !selectedId && (
        <>
          {tab === 'inbox' && (
            <div className="flex gap-2 mb-3">
              <select
                aria-label="Filter by status"
                value={statusFilter}
                onChange={e => setStatusFilter(e.target.value as StatusFilter)}
                className="rounded-lg border border-gray-200 px-2 py-1.5 text-sm"
              >
                <option value="open">Open</option>
                <option value="all">All</option>
                {(Object.keys(STATUS_LABELS) as FeedbackStatus[]).map(s => (
                  <option key={s} value={s}>{STATUS_LABELS[s]}</option>
                ))}
              </select>
              <select
                aria-label="Filter by category"
                value={categoryFilter}
                onChange={e => setCategoryFilter(e.target.value as 'all' | FeedbackCategory)}
                className="rounded-lg border border-gray-200 px-2 py-1.5 text-sm"
              >
                <option value="all">All categories</option>
                {CATEGORIES.map(c => <option key={c.value} value={c.value}>{c.label}</option>)}
              </select>
            </div>
          )}

          {visible.length === 0 ? (
            <p className="text-sm text-brown-muted">
              {tab === 'inbox' ? 'Nothing here.' : 'No feedback yet. Tap Feedback to send one.'}
            </p>
          ) : (
            <ul className="space-y-2">
              {visible.map(f => (
                <li key={f.id}>
                  <button
                    onClick={() => go({ id: f.id })}
                    className="w-full text-left bg-white rounded-xl p-3 shadow-sm hover:shadow transition-shadow"
                  >
                    <div className="flex items-center justify-between gap-2 mb-1">
                      <span className="text-xs font-semibold text-brown-dark">{CATEGORY_LABELS[f.category]}</span>
                      <span className="text-xs rounded-full bg-[#F5F0E8] px-2 py-0.5 text-brown-dark">{STATUS_LABELS[f.status]}</span>
                    </div>
                    <p className="text-sm text-brown-dark line-clamp-2">{f.message}</p>
                    <p className="text-xs text-brown-muted mt-1">
                      {tab === 'inbox' && <><span>{f.is_anonymous ? 'Anonymous' : f.submitter?.name ?? 'Unknown'}</span> · </>}
                      {formatWhen(f.created_at)}
                    </p>
                  </button>
                </li>
              ))}
            </ul>
          )}
        </>
      )}
    </div>
  )
}
```

```tsx
import { Component, ReactNode } from 'react'
import { openFeedback } from '../../features/feedback/feedbackBus'

interface Props { children: ReactNode }
interface State { error: Error | null }

export class ErrorBoundary extends Component<Props, State> {
  state: State = { error: null }

  static getDerivedStateFromError(error: Error): State {
    return { error }
  }

  render() {
    if (this.state.error) {
      const message = this.state.error.message
      return (
        <div className="min-h-[60vh] flex flex-col items-center justify-center px-4 text-center">
          <p className="text-2xl mb-2">⚠️</p>
          <p className="text-brown-dark font-semibold mb-1">Something went wrong</p>
          <p className="text-brown-muted text-sm mb-4">{message}</p>
          <div className="flex gap-2">
            <button
              onClick={() => this.setState({ error: null })}
              className="text-sm bg-brown-btn hover:bg-brown-btn-hover text-white px-4 py-2 rounded-lg transition-colors"
            >
              Try again
            </button>
            {/* 表单挂在 Sidebar 里,不在本 boundary 之内,所以这里崩了也打得开 */}
            <button
              onClick={() => openFeedback({ category: 'system', errorMessage: message })}
              className="text-sm border border-[#8B6344] text-brown-dark hover:bg-[#F5F0E8] px-4 py-2 rounded-lg transition-colors"
            >
              Report this problem
            </button>
          </div>
        </div>
      )
    }
    return this.props.children
  }
}
```
