-- 2026-10-23 — Org Chart: a narrow read of the reporting tree, plus a guard
-- against reporting loops.
--
-- get_org_chart(): every signed-in staff member can open the Org Chart, so it
-- needs every active person's name, rank, title, branch and supervisor. It
-- returns ONLY those columns, never IC number, phone, address or date of
-- birth, so the chart keeps working if staff_select is later narrowed to self
-- or same-branch, and so the page never pulls sensitive columns into the
-- browser.
--
-- staff_guard_supervisor_loop: staff_supervisor_not_self_check already stops
-- A → A. This also stops A → B → A (and longer loops), which would make a
-- chain that never reaches the owner. Only fires when supervisor_id actually
-- changes, so a profile save that resends an unchanged value never trips on
-- old data.
--
-- Idempotent: safe to re-run.

begin;

create or replace function get_org_chart()
returns table (
  id uuid,
  name text,
  nickname text,
  avatar text,
  rank text,
  system_role text,
  job_title text,
  branch text,
  department text,
  supervisor_id uuid
)
language sql
stable
security definer
set search_path = public
as $$
  select s.id, s.name::text, s.nickname::text, s.avatar::text, s.rank::text,
         s.system_role::text, r.name::text, s.branch::text, s.department::text,
         s.supervisor_id
  from staff s
  left join roles r on r.id = s.job_title_id
  where s.status = 'active'
    and auth.uid() is not null
  order by s.name;
$$;

revoke all on function get_org_chart() from public, anon;
grant execute on function get_org_chart() to authenticated;

create or replace function staff_guard_supervisor_loop()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_cur uuid := new.supervisor_id;
  v_steps int := 0;
begin
  if new.supervisor_id is null then
    return new;
  end if;
  if tg_op = 'UPDATE' and new.supervisor_id is not distinct from old.supervisor_id then
    return new;
  end if;
  -- Walk up from the new supervisor; reaching this row means a loop. Bounded
  -- so a loop already present in old data cannot hang the write.
  while v_cur is not null and v_steps < 100 loop
    if v_cur = new.id then
      raise exception 'Supervisor would create a reporting loop';
    end if;
    select s.supervisor_id into v_cur from staff s where s.id = v_cur;
    v_steps := v_steps + 1;
  end loop;
  return new;
end
$$;

drop trigger if exists staff_guard_supervisor_loop on staff;
create trigger staff_guard_supervisor_loop
  before insert or update of supervisor_id on staff
  for each row execute function staff_guard_supervisor_loop();

commit;

-- VERIFY 1 — the function's result has no sensitive columns, and anon cannot
-- call it (expect anon_can = false, auth_can = true):

--          has_function_privilege('anon', 'get_org_chart()', 'execute') as anon_can,

-- VERIFY 2 — the trigger rejects a two-step loop. Changes nothing: the final
-- raise rolls the whole block back; read the result from the error message
-- (expect "VERIFY_DONE: loop_blocked=true"):
--   do $$
--   declare a uuid; b uuid; a_old uuid; b_old uuid; blocked boolean := false;
--   begin
--     select id, supervisor_id into a, a_old from staff where status = 'active' order by name limit 1;
--     select id, supervisor_id into b, b_old from staff where status = 'active' and id <> a order by name limit 1;
--     update staff set supervisor_id = a where id = b;
--     begin
--       update staff set supervisor_id = b where id = a;
--     exception when others then
--       blocked := sqlerrm = 'Supervisor would create a reporting loop';

```ts
import type { Rank, SystemRole } from '../../shared/types'
import { RANK_ORDER } from '../../shared/types'

/** One row of get_org_chart(): only what the chart shows, never IC/phone/address. */
export interface OrgPerson {
  id: string
  name: string
  nickname: string | null
  avatar: string | null
  rank: Rank
  system_role: SystemRole
  job_title: string | null
  branch: string | null
  department: string | null
  supervisor_id: string | null
}

/** none: supervisor_id is null. inactive: it points at someone not in the
 *  active list. broken: has a supervisor, but the chain loops. */
export type UnassignedReason = 'none' | 'inactive' | 'broken'

export interface UnassignedEntry {
  person: OrgPerson
  reason: UnassignedReason
}

export interface OrgTree {
  byId: Map<string, OrgPerson>
  roots: OrgPerson[]
  /** Non-owners not reachable from an owner. Only the top of each detached
   *  chain is listed; people below it stay under it. */
  unassigned: UnassignedEntry[]
  childrenOf(id: string): OrgPerson[]
}

const isOwner = (p: OrgPerson) => p.system_role === 'owner'

/** Higher rank first, then name. */
export function compareStaff(a: OrgPerson, b: OrgPerson): number {
  const byRank = RANK_ORDER.indexOf(b.rank) - RANK_ORDER.indexOf(a.rank)
  return byRank !== 0 ? byRank : a.name.localeCompare(b.name)
}

export function buildOrgTree(rows: OrgPerson[]): OrgTree {
  const byId = new Map(rows.map(r => [r.id, r]))
  const children = new Map<string, OrgPerson[]>()
  for (const r of rows) {
    // An owner is always a root, even if someone set a supervisor on them.
    if (isOwner(r) || !r.supervisor_id || r.supervisor_id === r.id || !byId.has(r.supervisor_id)) continue
    const list = children.get(r.supervisor_id) ?? []
    list.push(r)
    children.set(r.supervisor_id, list)
  }
  for (const list of children.values()) list.sort(compareStaff)
  const childrenOf = (id: string) => children.get(id) ?? []

  const roots = rows.filter(isOwner).sort(compareStaff)

  const reached = new Set<string>()
  const queue = roots.map(r => r.id)
  while (queue.length) {
    const id = queue.shift()!
    if (reached.has(id)) continue
    reached.add(id)
    for (const c of childrenOf(id)) queue.push(c.id)
  }

  const unassigned: UnassignedEntry[] = []
  for (const r of rows) {
    if (isOwner(r) || reached.has(r.id)) continue
    if (!r.supervisor_id || r.supervisor_id === r.id) {
      unassigned.push({ person: r, reason: 'none' })
      continue
    }
    if (!byId.has(r.supervisor_id)) {
      unassigned.push({ person: r, reason: 'inactive' })
      continue
    }
    // Has a live supervisor but is unreached: walk up. If the walk ends at
    // someone with no valid supervisor, that person is the listed top, not r.
    // If it loops, list r only when r is part of the loop itself.
    // (Chains that reach an owner are already in `reached`.)
    let node: OrgPerson = r
    const seen: string[] = []
    let loopStart = -1
    for (;;) {
      seen.push(node.id)
      const next = node.supervisor_id ? byId.get(node.supervisor_id) : undefined
      if (!next || next.id === node.id) break          // ends at an unassigned top
      const at = seen.indexOf(next.id)
      if (at >= 0) { loopStart = at; break }           // found a loop
      node = next
    }
    if (loopStart >= 0 && seen.slice(loopStart).includes(r.id)) {
      unassigned.push({ person: r, reason: 'broken' })
    }
  }
  unassigned.sort((a, b) => a.person.name.localeCompare(b.person.name))

  return { byId, roots, unassigned, childrenOf }
}

/** Direct supervisor first, the top of the chain last. Stops at an owner, at
 *  a missing supervisor, or at the first repeat (bad data never loops). */
export function chainOf(tree: OrgTree, id: string): OrgPerson[] {
  const out: OrgPerson[] = []
  const seen = new Set<string>([id])
  let cur = tree.byId.get(id)
  while (cur && !isOwner(cur) && cur.supervisor_id) {
    const sup = tree.byId.get(cur.supervisor_id)
    if (!sup || seen.has(sup.id)) break
    out.push(sup)
    seen.add(sup.id)
    cur = sup
  }
  return out
}

export function descendantsOf(tree: OrgTree, id: string): Set<string> {
  const out = new Set<string>()
  const queue = tree.childrenOf(id).map(c => c.id)
  while (queue.length) {
    const next = queue.shift()!
    if (out.has(next) || next === id) continue
    out.add(next)
    for (const c of tree.childrenOf(next)) queue.push(c.id)
  }
  return out
}

/** Who may be picked as this person's supervisor: anyone active except the
 *  person and their own reports (picking one of those would make a loop). */
export function supervisorChoices(tree: OrgTree, personId: string): OrgPerson[] {
  const blocked = descendantsOf(tree, personId)
  blocked.add(personId)
  return [...tree.byId.values()].filter(p => !blocked.has(p.id)).sort(compareStaff)
}

/** Ids to expand so `levels` levels below each root are visible. */
export function initialExpanded(tree: OrgTree, levels: number): Set<string> {
  const out = new Set<string>()
  let frontier = tree.roots
  for (let depth = 0; depth < levels && frontier.length; depth++) {
    const next: OrgPerson[] = []
    for (const p of frontier) {
      if (out.has(p.id)) continue
      out.add(p.id)
      next.push(...tree.childrenOf(p.id))
    }
    frontier = next
  }
  return out
}

export function searchPeople(tree: OrgTree, query: string, limit = 8): OrgPerson[] {
  const q = query.trim().toLowerCase()
  if (!q) return []
  return [...tree.byId.values()]
    .filter(p => p.name.toLowerCase().includes(q) || (p.nickname ?? '').toLowerCase().includes(q))
    .sort((a, b) => a.name.localeCompare(b.name))
    .slice(0, limit)
}

import { supabase } from '../../shared/lib/supabase'
import { writeOutcome, type WriteOutcome, type WriteResponse } from '../../shared/lib/db-write'
import type { OrgPerson } from './orgTree'

/** Every active staff member, narrow columns only (see get_org_chart()). */
export async function fetchOrgChart(): Promise<OrgPerson[]> {
  const { data, error } = await supabase.rpc('get_org_chart')
  if (error) throw new Error(error.message)
  return (data ?? []) as OrgPerson[]
}

/** Sets one staff member's supervisor. The DB's loop guard rejects choices
 *  that would make a reporting loop; its message comes back as `reason`. */
export async function assignSupervisor(staffId: string, supervisorId: string): Promise<WriteOutcome> {
  const res = await supabase.from('staff').update({ supervisor_id: supervisorId }).eq('id', staffId).select('id')
  return writeOutcome(res as WriteResponse)
}

    meId,
    chainIds: new Set(chain.map(p => p.id)),
    highlightId,
    onOpen: canAssign ? (id: string) => navigate(`/staff/${id}`) : undefined,
  }
  const detachedTops = tree ? tree.unassigned.map(u => u.person) : []

  return (
    <div className="min-h-screen bg-[#F5F0E8]">
      <div className="max-w-6xl mx-auto px-4 py-6 lg:px-8">
        <PageHeader title="Org Chart" subtitle="Who reports to whom, from the owner down." />

        {loadError && (
          <Card className="mb-4">
            <p role="alert" className="text-sm text-bad">Couldn't load the org chart: {loadError}</p>
          </Card>
        )}

        {!tree || !ctx ? (
          !loadError && <p className="text-[#8B7355] animate-pulse">Loading org chart...</p>
        ) : (
          <>
            <ReportingLine me={me} chain={chain} />
            {canAssign && <UnassignedPanel tree={tree} onAssigned={load} />}

            <div className="relative mb-4 max-w-sm">
              <Search size={16} className="absolute left-3 top-2.5 text-brown-muted" aria-hidden />
              <input
                type="search"
                value={query}
                onChange={e => setQuery(e.target.value)}
                placeholder="Find a person…"
                aria-label="Find a person"
                className="w-full rounded-xl border border-border-mid bg-white py-2 pl-9 pr-3 text-sm text-brown-dark"
              />
              {matches.length > 0 && (
                <ul className="absolute z-10 mt-1 w-full rounded-xl bg-white py-1 shadow-card">
                  {matches.map(p => (
                    <li key={p.id}>
                      <button
                        type="button"
                        onClick={() => reveal(p.id)}
                        className="w-full px-3 py-2 text-left text-sm text-brown-dark hover:bg-cream"
                      >
                        {p.name} <span className="text-brown-muted">· {RANK_LABELS[p.rank]} · {shortenBranch(p.branch)}</span>
                      </button>
                    </li>
                  ))}
                </ul>
              )}
              {query.trim() && matches.length === 0 && (
                <p className="mt-1 text-xs text-brown-muted">No one matches “{query.trim()}”.</p>
              )}
            </div>

            {tree.roots.length === 0 && (
              <Card className="mb-4">
                <p className="text-sm text-brown-dark">
                  No owner is set yet. Give someone the Owner role in Settings so the chart has a top.
                </p>
              </Card>
            )}

            <div className="overflow-x-auto pb-4 -mx-4 px-4">
              <OrgTreeView roots={tree.roots} ctx={ctx} />
            </div>

            {detachedTops.length > 0 && (
              <section className="mt-8">
                <h2 className="mb-3 text-sm font-semibold text-brown-muted">Not connected to the owner yet</h2>
                <div className="overflow-x-auto pb-4 -mx-4 px-4">
                  <OrgTreeView roots={detachedTops} ctx={ctx} />
                </div>
              </section>
            )}
          </>
        )}
      </div>
    </div>
  )
}
```
